Attribute VB_Name = "modCompactJsonProbe"
Option Explicit
' ДЛЯ ТЕСТОВ: хранение вспомогательных таблиц в компактном JSON по столбцам.
' Заголовок записан один раз; постоянный столбец хранится одним значением.
' Числа не округляются до точности рисунка. Формат экспериментальный.
Private Const CHUNK_LENGTH As Long = 4000 ' Короткие фрагменты для надёжной пакетной записи через Excel Value2.
Private mJsonDecimalSeparator As String ' Разделитель для точного преобразования строки функцией CDbl.
Private Type TJsonDouble
    Value As Double
End Type
Private Type TJsonWords
    Low As Long
    High As Long
End Type

Public Function ProbeJsonEnabled() As Boolean
    ProbeJsonEnabled = (Left$(ProbeVariant, 4) = "Json")
End Function
Public Function ProbeJsonRange(ByVal rangeName As String) As Boolean
    If Not ProbeJsonEnabled Then Exit Function
    Select Case LCase$(rangeName)
        Case "rngndmsectiongeometry", "rngndmsectioncontours", "rngndmsectionannotations", "rngndmmaterialdiagrams"
            ProbeJsonRange = True
    End Select
End Function

' Единственное обращение записи массива; сериализация также входит в замер.
Public Sub ProbeWriteJson(ByVal target As Object, ByRef data As Variant)
    On Error GoTo Failed
    Dim stage As String: stage = "encode"
    Dim rows As Long, columns As Long, col As Long, row As Long, constant As Boolean
    Dim pieces() As String, values() As String, text As String, i As Long, count As Long
    rows = UBound(data, 1): columns = UBound(data, 2)
    ReDim pieces(1 To columns)
    For col = 1 To columns
        stage = "encode column " & CStr(col) & ", rows " & CStr(rows)
        constant = True
        For row = 3 To rows
            If VarType(data(row, col)) <> VarType(data(2, col)) Then constant = False: Exit For
            If data(row, col) <> data(2, col) Then constant = False: Exit For
        Next row
        If rows = 1 Then
            pieces(col) = "[" & JsonToken(data(1, col)) & ",0,null]"
        ElseIf constant Then
            pieces(col) = "[" & JsonToken(data(1, col)) & ",0," & JsonToken(data(2, col)) & "]"
        Else
            ReDim values(1 To rows - 1)
            For row = 2 To rows: values(row - 1) = JsonToken(data(row, col)): Next row
            pieces(col) = "[" & JsonToken(data(1, col)) & ",1,[" & Join(values, ",") & "]]"
        End If
    Next col
    text = "[" & CStr(rows) & "," & CStr(columns) & ",[" & Join(pieces, ",") & "]]"
    count = (Len(text) + CHUNK_LENGTH - 1) \ CHUNK_LENGTH
    stage = "write " & CStr(Len(text)) & " characters / " & CStr(count) & " chunks"
    Dim output() As Variant: ReDim output(1 To count + 1, 1 To 5)
    output(1, 1) = "JSON experiment": output(1, 2) = count
    output(1, 3) = rows: output(1, 4) = columns
    If rows > 1 And CStr(data(1, 1)) = "RunID" Then output(1, 5) = CDbl(data(2, 1))
    For i = 1 To count
        output(i + 1, 1) = i: output(i + 1, 2) = Mid$(text, (i - 1) * CHUNK_LENGTH + 1, CHUNK_LENGTH)
    Next i
    With target.Resize(count + 1, 5)
        .NumberFormat = "General": .WrapText = False
    End With
    With target.Offset(1, 1).Resize(count, 1)
        .NumberFormat = "@": .HorizontalAlignment = xlRight
    End With
    stage = stage & ", Value2"
    target.Resize(count + 1, 5).Value2 = output
    Exit Sub
Failed:
    Err.Raise Err.Number, "ProbeWriteJson", stage & ": " & Err.Description
End Sub

' Читает только хранимый JSON из Excel. Готовые модели между чтениями не кэшируются.
Public Function ProbeReadJson(ByVal workbook As Object, ByVal rangeName As String) As Variant
    On Error GoTo Failed
    Dim stage As String: stage = "read metadata " & rangeName
    mJsonDecimalSeparator = Mid$(CStr(0.5), 2, 1)
    Dim target As Object, metadata As Variant, chunks As Variant, parts() As String, text As String
    Set target = workbook.Names.Item(rangeName).RefersToRange
    metadata = target.Resize(1, 5).Value2
    If CStr(metadata(1, 1)) <> "JSON experiment" Then Exit Function
    Dim count As Long, i As Long, position As Long, rows As Long, columns As Long, col As Long, row As Long
    count = CLng(metadata(1, 2))
    chunks = target.Offset(1, 1).Resize(count, 1).Value2
    ReDim parts(1 To count)
    If count = 1 Then
        parts(1) = CStr(chunks)
    Else
        For i = 1 To count: parts(i) = CStr(chunks(i, 1)): Next i
    End If
    text = Join(parts, ""): position = 1
    stage = "parse " & rangeName & ", " & CStr(Len(text)) & " characters"
    JsonExpect text, position, "[": rows = CLng(JsonReadValue(text, position))
    JsonExpect text, position, ",": columns = CLng(JsonReadValue(text, position))
    If rows <> CLng(metadata(1, 3)) Or columns <> CLng(metadata(1, 4)) Then Err.Raise 5, , "JSON dimensions differ"
    Dim data() As Variant, flag As Long, value As Variant
    ReDim data(1 To rows, 1 To columns)
    JsonExpect text, position, ",": JsonExpect text, position, "["
    For col = 1 To columns
        stage = "parse " & rangeName & " column " & CStr(col)
        If col > 1 Then JsonExpect text, position, ","
        JsonExpect text, position, "[": data(1, col) = JsonReadValue(text, position)
        JsonExpect text, position, ",": flag = CLng(JsonReadValue(text, position))
        JsonExpect text, position, ","
        If flag = 0 Then
            value = JsonReadValue(text, position)
            For row = 2 To rows: data(row, col) = value: Next row
        ElseIf flag = 1 Then
            JsonExpect text, position, "["
            For row = 2 To rows
                If row > 2 Then JsonExpect text, position, ","
                data(row, col) = JsonReadValue(text, position)
            Next row
            JsonExpect text, position, "]"
        Else
            Err.Raise 5, , "Unknown JSON column encoding"
        End If
        JsonExpect text, position, "]"
    Next col
    JsonExpect text, position, "]": JsonExpect text, position, "]"
    If position <> Len(text) + 1 Then Err.Raise 5, , "Unexpected JSON suffix"
    If CStr(data(1, 1)) = "RunID" And Not IsEmpty(metadata(1, 5)) Then
        For row = 2 To rows: data(row, 1) = CDbl(metadata(1, 5)): Next row
    End If
    ProbeReadJson = data
    Exit Function
Failed:
    Err.Raise Err.Number, "ProbeReadJson", stage & ": " & Err.Description
End Function

' Дата запуска меняется одной ячейкой без перепаковки всей геометрии.
Public Sub ProbeJsonRunId(ByVal workbook As Object, ByVal runID As Date)
    Dim name As Variant
    For Each name In Array("rngNDMSectionGeometry", "rngNDMSectionContours")
        workbook.Names.Item(CStr(name)).RefersToRange.Offset(0, 4).Value2 = CDbl(runID)
    Next name
End Sub

Private Function JsonToken(ByVal value As Variant) As String
    Select Case VarType(value)
        Case vbEmpty, vbNull: JsonToken = "null"
        Case vbBoolean: JsonToken = IIf(CBool(value), "true", "false")
        Case vbString
            If Left$(CStr(value), 1) = "~" Then value = "~" & CStr(value)
            JsonToken = JsonQuote(CStr(value))
        Case Else
            If CDbl(value) = Fix(CDbl(value)) And Abs(CDbl(value)) < 1000000000000000# Then
                JsonToken = Trim$(Str$(CDbl(value)))
            ElseIf InStr(ProbeVariant, "Exact") > 0 Then
                Dim scalar As TJsonDouble, words As TJsonWords
                scalar.Value = CDbl(value): LSet words = scalar
                JsonToken = """~D" & Right$("00000000" & Hex$(words.High), 8) & Right$("00000000" & Hex$(words.Low), 8) & """"
            Else
                JsonToken = SectionOptionalNumber(CDbl(value))
            End If
    End Select
End Function
Private Function JsonQuote(ByVal value As String) As String
    value = Replace$(value, "\", "\\")
    value = Replace$(value, """", "\""")
    Dim i As Long
    For i = 0 To 31
        If InStr(value, Chr$(i)) > 0 Then value = Replace$(value, Chr$(i), "\u" & Right$("0000" & Hex$(i), 4))
    Next i
    JsonQuote = """" & value & """"
End Function
Private Sub JsonExpect(ByRef text As String, ByRef position As Long, ByVal expected As String)
    If Mid$(text, position, 1) <> expected Then Err.Raise 5, , "Unexpected JSON token at " & CStr(position)
    position = position + 1
End Sub
Private Function JsonReadValue(ByRef text As String, ByRef position As Long) As Variant
    Dim token As String, start As Long, finish As Long, slashes As Long, character As String
    start = position: character = Mid$(text, position, 1)
    If character = """" Then
        position = position + 1: start = position
        Do
            finish = InStr(position, text, """")
            If finish = 0 Then Err.Raise 5, , "Unterminated JSON string"
            slashes = 0
            Do While finish - slashes > start
                If Mid$(text, finish - slashes - 1, 1) <> "\" Then Exit Do
                slashes = slashes + 1
            Loop
            If slashes Mod 2 = 0 Then Exit Do
            position = finish + 1
        Loop
        token = Mid$(text, start, finish - start): position = finish + 1
        token = JsonUnescape(token)
        If Left$(token, 2) = "~D" And Len(token) = 18 Then
            Dim words As TJsonWords, scalar As TJsonDouble
            words.High = CLng("&H" & Mid$(token, 3, 8)): words.Low = CLng("&H" & Mid$(token, 11, 8))
            LSet scalar = words: JsonReadValue = scalar.Value
        ElseIf Left$(token, 2) = "~~" Then
            JsonReadValue = Mid$(token, 2)
        Else
            JsonReadValue = token
        End If
        Exit Function
    End If
    Do While position <= Len(text)
        character = Mid$(text, position, 1)
        If character = "," Or character = "]" Then Exit Do
        position = position + 1
    Loop
    token = Mid$(text, start, position - start)
    Select Case token
        Case "null": JsonReadValue = Empty
        Case "true": JsonReadValue = True
        Case "false": JsonReadValue = False
        Case Else
            If Len(token) = 0 Then Err.Raise 5, , "Empty JSON number"
            If mJsonDecimalSeparator <> "." Then token = Replace$(token, ".", mJsonDecimalSeparator)
            JsonReadValue = CDbl(token)
    End Select
End Function

' Сохраняет фактические JSON-фрагменты для независимой проверки синтаксиса вне Excel.
' Вызывается после остановки таймера, один раз для выбранного режима и примера.
Public Sub ProbeExportJson(ByVal workbook As Object, ByVal fixture As String)
    Dim name As Variant, target As Object, metadata As Variant, chunks As Variant, parts() As String
    Dim count As Long, i As Long, text As String, stream As Object
    For Each name In Array("rngNDMSectionGeometry", "rngNDMSectionContours", "rngNDMSectionAnnotations", "rngNDMMaterialDiagrams")
        Set target = workbook.Names.Item(CStr(name)).RefersToRange
        metadata = target.Resize(1, 5).Value2: count = CLng(metadata(1, 2))
        chunks = target.Offset(1, 1).Resize(count, 1).Value2
        ReDim parts(1 To count)
        If count = 1 Then
            parts(1) = CStr(chunks)
        Else
            For i = 1 To count: parts(i) = CStr(chunks(i, 1)): Next i
        End If
        text = Join(parts, "")
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 2: stream.Charset = "utf-8": stream.Open
        stream.WriteText text
        stream.SaveToFile ProbeDirectory & "\" & fixture & "_" & ProbeVariant & "_" & CStr(name) & ".json", 2
        stream.Close
        ProbeLog fixture & "_JsonSize.csv", CStr(name) & "," & CStr(metadata(1, 3)) & "," & CStr(metadata(1, 4)) & "," & CStr(count) & "," & CStr(Len(text))
    Next name
End Sub
Private Function JsonUnescape(ByVal text As String) As String
    If InStr(text, "\") = 0 Then JsonUnescape = text: Exit Function
    Dim i As Long, character As String, result As String, code As Long
    i = 1
    Do While i <= Len(text)
        character = Mid$(text, i, 1): i = i + 1
        If character = "\" Then
            character = Mid$(text, i, 1): i = i + 1
            Select Case character
                Case "u"
                    code = CLng("&H" & Mid$(text, i, 4)): i = i + 4
                    If code > 32767 Then code = code - 65536
                    character = ChrW$(code)
                Case """", "\", "/"
                Case "n": character = vbLf
                Case "r": character = vbCr
                Case "t": character = vbTab
                Case "b": character = Chr$(8)
                Case "f": character = Chr$(12)
                Case Else: Err.Raise 5, , "Unknown JSON escape"
            End Select
        End If
        result = result & character
    Loop
    JsonUnescape = result
End Function
